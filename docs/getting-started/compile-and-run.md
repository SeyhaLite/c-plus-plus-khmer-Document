# ការបកប្រែ និងការប្រតិបត្តិកម្មវិធី

```{translation-meta} getting-started/compile-and-run
```

ការសរសេរកូដគឺជាជំហានទីមួយ។ ដើម្បីទទួលបានកម្មវិធីដែលដំណើរការ កូដត្រូវឆ្លងកាត់
ការបកប្រែ (compilation) និងការភ្ជាប់ (linking)។

## ពាក្យបញ្ជាមូលដ្ឋាន

```bash
g++ -std=c++17 hello.cpp -o hello
./hello
```

- `g++` ជាកម្មវិធីបកប្រែ C++។
- `-std=c++17` ជ្រើសស្តង់ដារ C++17។
- `hello.cpp` ជាឯកសារប្រភព។
- `-o hello` កំណត់ឈ្មោះឯកសារលទ្ធផល។ បើមិនកំណត់ វានឹងបង្កើត `a.out`។
- `./hello` ដំណើរការកម្មវិធី (នៅ Linux/macOS)។ នៅ Windows គេរត់ `hello.exe`។

## ដំណាក់កាលនៃការបកប្រែ

តាមស្តង់ដារ C++ ការបកប្រែឆ្លងកាត់ដំណាក់កាលជាច្រើន (translation phases)។
ជាក់ស្តែង អ្នកអាចគិតថាវាមានបីជំហានធំៗ៖

1. **ព្រៃដំណើរការ (preprocessing)** — ដោះស្រាយ `#include`, `#define`
   និងលក្ខខណ្ឌ `#if`។
2. **បកប្រែ (compilation)** — បំប្លែងកូដ C++ ទៅជាកូដម៉ាស៊ីន ឬកូដ object។
   នៅជំហាននេះ កម្មវិធីបកប្រែចាប់កំហុសវាក្យសម្ព័ន្ធ និងប្រភេទ។
3. **ភ្ជាប់ (linking)** — ភ្ជាប់កូដ object ជាមួយបណ្ណាល័យ ដើម្បីបង្កើតកម្មវិធីពេញលេញ។

បើអ្នកចង់ឃើញជំហានទាំងនេះដោយឡែក៖

```bash
g++ -std=c++17 -c hello.cpp -o hello.o   # បកប្រែជា object មិនភ្ជាប់
g++ hello.o -o hello                     # ភ្ជាប់
```

## កូដឧទាហរណ៍៖ ពេលបកប្រែ និងពេលដំណើរការ

```{literalinclude} ../../examples/getting-started/compile-and-run.cpp
:language: cpp
```

```bash
g++ -std=c++17 -Wall -Wextra compile-and-run.cpp -o demo
./demo
```

លទ្ធផល៖

```text
compile-time: 5
runtime: 9
```

`constexpr` បញ្ជាឱ្យកម្មវិធីបកប្រែគណនាតម្លៃនៅពេលបកប្រែ បើអាចធ្វើបាន។
ដូច្នេះ `compile_time` មិនចំណាយពេលគណនានៅពេលកម្មវិធីដំណើរការទេ។
ផ្ទុយទៅវិញ `add(4, 5)` ត្រូវបានគណនានៅពេលដំណើរការ។

:::{note}
ការគណនានៅពេលបកប្រែមិនមែនជាការធានាថាតម្លៃនឹងត្រូវគណនាមុននោះទេ។
ស្តង់ដារអនុញ្ញាតឱ្យកម្មវិធីបកប្រែជ្រើសរើសវិធី ដរាបណាលទ្ធផលត្រឹមត្រូវ។
:::

## កូដត្រឡប់ (exit code)

កម្មវិធីត្រឡប់លេខទៅប្រព័ន្ធប្រតិបត្តិការ។ នៅ shell អ្នកអាចពិនិត្យវាបាន៖

```bash
./demo
echo $?
```

បើកម្មវិធីត្រឡប់ `0` នោះមានន័យថាជោគជ័យ។ លេខផ្សេងទៀតជាធម្មតាបង្ហាញកំហុស។
នេះសំខាន់ព្រោះឧបករណ៍ស្វ័យប្រវត្តិ (CI) ពឹងលើលេខនេះដើម្បីដឹងថាការធ្វើតេស្តជោគជ័យ។

## ជម្រើសដែលគួរប្រើញឹកញាប់

| ជម្រើស | អត្ថន័យ |
| --- | --- |
| `-Wall -Wextra` | បើកការព្រមានសំខាន់ៗ |
| `-Werror` | ចាត់ទុកការព្រមានជាកំហុស |
| `-O2` | បង្កើនប្រសិទ្ធភាព |
| `-g` | បន្ថែមព័ត៌មាន debug |
| `-std=c++20` | ជ្រើសស្តង់ដារ C++20 |

សម្រាប់ការរៀន សូមប្រើ `-Wall -Wextra` ជានិច្ច។ វាជួយអ្នកឃើញបញ្ហាមុនពេលវាក្លាយជា bug។
